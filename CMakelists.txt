# ============================================================
#  Compiler options (MSVC)
# ============================================================
if(MSVC)
    target_compile_options(${PROJECT_NAME} PRIVATE
        /MP                 # параллельная компиляция
        /W3                 # предупреждения уровня 3
        /wd4996             # отключить warning deprecated CRT
        /wd4244 /wd4267     # truncation int/size_t (RakNet полон этим)
        $<$<CONFIG:Release>:/O2 /GL /Gy>   # максимальная оптимизация
        $<$<CONFIG:Debug>:/Od /RTC1 /Zi>   # отладка
    )
    target_link_options(${PROJECT_NAME} PRIVATE
        $<$<CONFIG:Release>:/LTCG /OPT:REF /OPT:ICF>
    )
    # Кодировка исходников — MultiByte (как в vcxproj)
    # CMake сам не пробрасывает CharacterSet, но /D_MBCS делает это явно
    target_compile_definitions(${PROJECT_NAME} PRIVATE _MBCS)
endif()
